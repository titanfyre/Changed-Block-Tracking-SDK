cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ChangedBlockTracking VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ on Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument user mode" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(ChangedBlocks SDK/ChangedBlocks.cpp SDK/Inventory.cpp SDK/License.cpp)
target_link_libraries(ChangedBlocks PUBLIC setupapi cfgmgr32 winhttp crypt32 advapi32)
add_executable(ChangedBlocksTests Tests/Protocol.cpp)
target_link_libraries(ChangedBlocksTests PRIVATE ChangedBlocks)
enable_testing()
add_test(NAME BitmapProtocolAndRanges COMMAND ChangedBlocksTests)
add_library(ChangedBlocksSetupCore InstallerUI/Setup.cpp InstallerUI/DriverSetup.cpp InstallerUI/RuntimeSetup.cpp)
target_link_libraries(ChangedBlocksSetupCore PUBLIC ChangedBlocks setupapi newdev wintrust crypt32 advapi32 ole32 shell32)
add_executable(ChangedBlocksInstallerTests Tests/Installer.cpp)
target_link_libraries(ChangedBlocksInstallerTests PRIVATE ChangedBlocksSetupCore)
add_test(NAME InstallerIdentityAndTrustRefusal COMMAND ChangedBlocksInstallerTests)
add_executable(ChangedBlocksStudio WIN32 App/Main.cpp)
target_link_libraries(ChangedBlocksStudio PRIVATE ChangedBlocks user32 gdi32 comctl32 comdlg32 shell32 ole32)
add_executable(ChangedBlocksCLI App/Console.cpp)
target_link_libraries(ChangedBlocksCLI PRIVATE ChangedBlocks)
add_executable(ChangedBlocksExample Examples/track.cpp)
target_link_libraries(ChangedBlocksExample PRIVATE ChangedBlocks)
enable_language(RC)
set(RUNTIME_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed runtime package")
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed CBT package")
set(resources "")
foreach(kind RUNTIME DRIVER)
 if(${kind}_PACKAGE_DIRECTORY)
  file(TO_CMAKE_PATH "${${kind}_PACKAGE_DIRECTORY}" directory)
  if(kind STREQUAL "RUNTIME")
   set(prefix TitanSdkRuntime)
   set(base 101)
  else()
   set(prefix TitanChangedBlockTracking)
   set(base 111)
  endif()
  foreach(extension inf cat sys)
   if(NOT EXISTS "${directory}/${prefix}.${extension}")
    message(FATAL_ERROR "Incomplete opaque ${kind} package")
   endif()
   string(APPEND resources "${base} RCDATA \"${directory}/${prefix}.${extension}\"\n")
   math(EXPR base "${base}+1")
  endforeach()
 endif()
endforeach()
set(package "${CMAKE_CURRENT_BINARY_DIR}/AppPackage-$<CONFIG>.rc")
file(GENERATE OUTPUT "${package}" CONTENT "${resources}201 RCDATA \"$<TARGET_FILE:ChangedBlocksStudio>\"\n202 RCDATA \"$<TARGET_FILE:ChangedBlocksCLI>\"\n203 RCDATA \"$<TARGET_FILE:ChangedBlocksExample>\"\n")
add_executable(ChangedBlocksSetup WIN32 InstallerUI/Main.cpp "${package}")
add_executable(ChangedBlocksSetupConsole InstallerConsole/Main.cpp "${package}")
foreach(target ChangedBlocksSetup ChangedBlocksSetupConsole)
 target_link_libraries(${target} PRIVATE ChangedBlocksSetupCore user32 gdi32 comctl32)
 add_dependencies(${target} ChangedBlocksStudio ChangedBlocksCLI ChangedBlocksExample)
endforeach()
foreach(target ChangedBlocksStudio ChangedBlocksCLI ChangedBlocksExample ChangedBlocksSetup ChangedBlocksSetupConsole)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target ChangedBlocksStudio ChangedBlocksCLI ChangedBlocksExample ChangedBlocksSetup ChangedBlocksSetupConsole ChangedBlocksTests ChangedBlocksInstallerTests)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
